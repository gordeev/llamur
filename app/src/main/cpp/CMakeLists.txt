cmake_minimum_required(VERSION 3.31.6)
project(motoai-probe LANGUAGES CXX)
# Keep __FILE__ and debug paths independent of the checkout location.
get_filename_component(MOTOAI_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${MOTOAI_SOURCE_ROOT}=."
                    "-ffile-prefix-map=${CMAKE_ANDROID_NDK}=/android-ndk")
add_library(motoai-probe SHARED probe.cpp)
target_compile_features(motoai-probe PRIVATE cxx_std_17)
target_compile_options(motoai-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(motoai-probe PRIVATE dl vulkan)

set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
option(MOTOAI_CPU_KLEIDIAI "Compile optional KleidiAI buffers (runtime native policy still disables extras)" ON)
set(GGML_CPU_KLEIDIAI ${MOTOAI_CPU_KLEIDIAI} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(MOTOAI_GPU_BACKEND "none" CACHE STRING "Explicit experimental backend; none keeps CPU only")
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
if(MOTOAI_GPU_BACKEND STREQUAL "vulkan" OR MOTOAI_GPU_BACKEND STREQUAL "both")
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    # Pinned upstream finds the SPIRV config but does not attach its target includes.
    include_directories("${MOTOAI_SPIRV_INCLUDE}")
endif()
if(MOTOAI_GPU_BACKEND STREQUAL "opencl" OR MOTOAI_GPU_BACKEND STREQUAL "both")
    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
    set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
    set(GGML_OPENCL_USE_ADRENO_BIN_KERNELS OFF CACHE BOOL "" FORCE)
endif()
if(NOT MOTOAI_GPU_BACKEND MATCHES "^(none|vulkan|opencl|both)$")
    message(FATAL_ERROR "Unknown MotoAI GPU backend")
endif()
add_subdirectory(../../../../native-runtime runtime-build)
add_library(motoai-runtime SHARED runtime-jni.cpp)
target_include_directories(motoai-runtime PRIVATE ../../../../third_party/llama.cpp/vendor)
target_link_libraries(motoai-runtime PRIVATE motoai-core)
add_library(motoai-checks SHARED backend-checks-jni.cpp ../../../../native-runtime/tests/numeric-cases.cpp)
target_include_directories(motoai-checks PRIVATE ../../../../native-runtime/tests "${MOTOAI_LLAMA_SOURCE}/ggml/src")
target_link_libraries(motoai-checks PRIVATE motoai-core)
