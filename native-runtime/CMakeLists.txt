cmake_minimum_required(VERSION 3.31.6)
project(motoai-runtime LANGUAGES C CXX)
set(MOTOAI_LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp" CACHE PATH "Pinned source or explicitly prepared experimental patch checkout")
set(MOTOAI_RUNTIME_VARIANT "baseline" CACHE STRING "Explicit runtime patch variant identifier")
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
# Source archives must not inherit the enclosing app's Git revision or history depth.
set(ENV{GIT_CEILING_DIRECTORIES} "${MOTOAI_LLAMA_SOURCE}")
set(MOTOAI_UPSTREAM_COMMIT "" CACHE STRING "Explicit prepared source revision; empty follows the baseline upstream lock")
if(NOT MOTOAI_UPSTREAM_COMMIT)
    # Match the selected source to a known pristine/patch pin; never label a new
    # engine with the historical baseline revision by accident.
    file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.." MOTOAI_REPO_ROOT)
    file(REAL_PATH "${MOTOAI_LLAMA_SOURCE}" MOTOAI_SELECTED_SOURCE)
    set(MOTOAI_SELECTED_LOCK "")
    foreach(MOTOAI_LOCK_NAME upstream.lock.json upstream-v0.9.0.lock.json)
        file(READ "${MOTOAI_REPO_ROOT}/${MOTOAI_LOCK_NAME}" MOTOAI_LOCK_DATA)
        string(JSON MOTOAI_RAW_PATH GET "${MOTOAI_LOCK_DATA}" path)
        file(REAL_PATH "${MOTOAI_REPO_ROOT}/${MOTOAI_RAW_PATH}" MOTOAI_RAW_SOURCE)
        if(MOTOAI_SELECTED_SOURCE STREQUAL MOTOAI_RAW_SOURCE)
            set(MOTOAI_SELECTED_LOCK "${MOTOAI_LOCK_NAME}")
        endif()
    endforeach()
    file(READ "${MOTOAI_REPO_ROOT}/patches/cpu-numeric/series.json" MOTOAI_SERIES)
    string(JSON MOTOAI_VARIANT_COUNT LENGTH "${MOTOAI_SERIES}" variants)
    math(EXPR MOTOAI_VARIANT_LAST "${MOTOAI_VARIANT_COUNT}-1")
    foreach(MOTOAI_INDEX RANGE ${MOTOAI_VARIANT_LAST})
        string(JSON MOTOAI_VARIANT_NAME MEMBER "${MOTOAI_SERIES}" variants ${MOTOAI_INDEX})
        string(JSON MOTOAI_VARIANT_PATH GET "${MOTOAI_SERIES}" variants "${MOTOAI_VARIANT_NAME}" directory)
        file(REAL_PATH "${MOTOAI_REPO_ROOT}/${MOTOAI_VARIANT_PATH}" MOTOAI_VARIANT_SOURCE)
        if(MOTOAI_SELECTED_SOURCE STREQUAL MOTOAI_VARIANT_SOURCE)
            string(JSON MOTOAI_SELECTED_LOCK ERROR_VARIABLE MOTOAI_LOCK_ERROR GET "${MOTOAI_SERIES}" variants "${MOTOAI_VARIANT_NAME}" upstream_lock)
            if(MOTOAI_LOCK_ERROR)
                set(MOTOAI_SELECTED_LOCK "upstream.lock.json")
            endif()
        endif()
    endforeach()
    if(NOT MOTOAI_SELECTED_LOCK)
        message(FATAL_ERROR "Custom engine sources require an explicit MOTOAI_UPSTREAM_COMMIT")
    endif()
    file(READ "${MOTOAI_REPO_ROOT}/${MOTOAI_SELECTED_LOCK}" MOTOAI_UPSTREAM_LOCK)
    string(JSON MOTOAI_UPSTREAM_COMMIT GET "${MOTOAI_UPSTREAM_LOCK}" commit)
endif()
set(LLAMA_BUILD_COMMIT "${MOTOAI_UPSTREAM_COMMIT}")
set(LLAMA_BUILD_NUMBER 0)
if(NOT TARGET llama)
    add_subdirectory("${MOTOAI_LLAMA_SOURCE}" llama-build)
endif()
add_library(motoai-core STATIC runtime.cpp protocol.cpp cpu-scheduling.cpp)
target_compile_definitions(motoai-core PRIVATE MOTOAI_RUNTIME_VARIANT="${MOTOAI_RUNTIME_VARIANT}")
# Numerical/API mechanics do not qualify real-model prefix-state reuse.
if(MOTOAI_UPSTREAM_COMMIT STREQUAL "701fd25b3a55a90297dec0d925204b29084054c8")
    target_compile_definitions(motoai-core PUBLIC MOTOAI_PREFIX_CACHE_QUALIFIED=1)
else()
    target_compile_definitions(motoai-core PUBLIC MOTOAI_PREFIX_CACHE_QUALIFIED=0)
endif()
set_target_properties(motoai-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(motoai-core PUBLIC cxx_std_17)
target_include_directories(motoai-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${MOTOAI_LLAMA_SOURCE}/vendor)
target_include_directories(motoai-core PRIVATE ${MOTOAI_LLAMA_SOURCE}/src)
target_link_libraries(motoai-core PUBLIC llama llama-common)
add_executable(motoai-backend-ops EXCLUDE_FROM_ALL ${MOTOAI_LLAMA_SOURCE}/tests/test-backend-ops.cpp)
target_compile_features(motoai-backend-ops PRIVATE cxx_std_17)
target_link_libraries(motoai-backend-ops PRIVATE ggml)
add_executable(motoai-numeric EXCLUDE_FROM_ALL tests/numeric-test.cpp tests/numeric-cases.cpp)
target_link_libraries(motoai-numeric PRIVATE motoai-core)
target_include_directories(motoai-numeric PRIVATE ${MOTOAI_LLAMA_SOURCE}/ggml/src)
file(STRINGS "${MOTOAI_LLAMA_SOURCE}/ggml/include/ggml.h" MOTOAI_INDEXED_ATTENTION_API REGEX "ggml_flash_attn_ext_rows")
if(MOTOAI_INDEXED_ATTENTION_API)
    add_executable(motoai-attention-rows-test EXCLUDE_FROM_ALL tests/attention-rows-test.cpp)
    target_link_libraries(motoai-attention-rows-test PRIVATE motoai-core)

target_link_libraries(motoai-load-plan PRIVATE motoai-core)

target_include_directories(motoai-repack-audit PRIVATE ${MOTOAI_LLAMA_SOURCE}/ggml/src)
target_link_libraries(motoai-repack-audit PRIVATE motoai-core)

target_compile_features(motoai-prefill-pack-bench PRIVATE cxx_std_17)
target_include_directories(motoai-prefill-pack-bench PRIVATE ${MOTOAI_LLAMA_SOURCE}/ggml/src ${MOTOAI_LLAMA_SOURCE}/vendor)
