cmake_minimum_required(VERSION 3.31.6)
project(motoai-runtime LANGUAGES C CXX)
set(MOTOAI_LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp" CACHE PATH "Pinned source or explicitly prepared experimental patch checkout")
set(MOTOAI_RUNTIME_VARIANT "baseline" CACHE STRING "Explicit runtime patch variant identifier")
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
# Source archives must not inherit the enclosing app's Git revision or history depth.
set(ENV{GIT_CEILING_DIRECTORIES} "${MOTOAI_LLAMA_SOURCE}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../upstream.lock.json" MOTOAI_UPSTREAM_LOCK)
string(JSON MOTOAI_UPSTREAM_COMMIT GET "${MOTOAI_UPSTREAM_LOCK}" commit)
set(LLAMA_BUILD_COMMIT "${MOTOAI_UPSTREAM_COMMIT}")
set(LLAMA_BUILD_NUMBER 0)
if(NOT TARGET llama)
    add_subdirectory("${MOTOAI_LLAMA_SOURCE}" llama-build)
endif()
add_library(motoai-core STATIC runtime.cpp protocol.cpp cpu-scheduling.cpp)
target_compile_definitions(motoai-core PRIVATE MOTOAI_RUNTIME_VARIANT="${MOTOAI_RUNTIME_VARIANT}")
set_target_properties(motoai-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(motoai-core PUBLIC cxx_std_17)
target_include_directories(motoai-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp/vendor)
target_include_directories(motoai-core PRIVATE ${MOTOAI_LLAMA_SOURCE}/src)
target_link_libraries(motoai-core PUBLIC llama llama-common)
add_executable(motoai-backend-ops EXCLUDE_FROM_ALL ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp/tests/test-backend-ops.cpp)
target_compile_features(motoai-backend-ops PRIVATE cxx_std_17)
target_link_libraries(motoai-backend-ops PRIVATE ggml)
add_executable(motoai-numeric EXCLUDE_FROM_ALL tests/numeric-test.cpp tests/numeric-cases.cpp)
target_link_libraries(motoai-numeric PRIVATE motoai-core)
target_include_directories(motoai-numeric PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp/ggml/src)
add_executable(motoai-swa-check EXCLUDE_FROM_ALL tools/swa-check.cpp)
target_link_libraries(motoai-swa-check PRIVATE motoai-core)
add_executable(motoai-state-prefix-check EXCLUDE_FROM_ALL tools/state-prefix-check.cpp)
add_executable(motoai-prefix-cache-test EXCLUDE_FROM_ALL tests/prefix-cache-test.cpp)
target_link_libraries(motoai-prefix-cache-test PRIVATE motoai-core)
target_link_libraries(motoai-state-prefix-check PRIVATE motoai-core)
target_include_directories(motoai-state-prefix-check PRIVATE ${MOTOAI_LLAMA_SOURCE}/src)
add_executable(motoai-tokenizer-tests EXCLUDE_FROM_ALL tests/tokenizer-test.cpp)
target_link_libraries(motoai-tokenizer-tests PRIVATE motoai-core)
add_executable(motoai-math EXCLUDE_FROM_ALL tools/math-probe.cpp)
add_executable(motoai-load-plan EXCLUDE_FROM_ALL tools/load-plan.cpp)
target_include_directories(motoai-load-plan PRIVATE ${MOTOAI_LLAMA_SOURCE}/src ${MOTOAI_LLAMA_SOURCE}/ggml/src)
target_link_libraries(motoai-load-plan PRIVATE motoai-core)
add_executable(motoai-repack-audit EXCLUDE_FROM_ALL tests/repack-audit.cpp)
target_include_directories(motoai-repack-audit PRIVATE ${MOTOAI_LLAMA_SOURCE}/ggml/src)
target_link_libraries(motoai-repack-audit PRIVATE motoai-core)
add_executable(motoai-prefill-pack-bench EXCLUDE_FROM_ALL tools/prefill-pack-bench.cpp)
target_compile_features(motoai-prefill-pack-bench PRIVATE cxx_std_17)
target_include_directories(motoai-prefill-pack-bench PRIVATE ${MOTOAI_LLAMA_SOURCE}/ggml/src ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp/vendor)
target_link_libraries(motoai-prefill-pack-bench PRIVATE ggml ${CMAKE_DL_LIBS})
add_executable(motoai-attention EXCLUDE_FROM_ALL tools/attention-replay.cpp)
target_link_libraries(motoai-attention PRIVATE motoai-core ${CMAKE_DL_LIBS})
target_compile_features(motoai-math PRIVATE cxx_std_17)
target_include_directories(motoai-math PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/llama.cpp/vendor)
target_link_libraries(motoai-math PRIVATE ${CMAKE_DL_LIBS})
if(NOT ANDROID)
    enable_testing()
    add_executable(runtime-run tools/runtime-run.cpp)
    target_link_libraries(runtime-run PRIVATE motoai-core)
    add_executable(runtime-test tests/runtime-test.cpp)
    target_link_libraries(runtime-test PRIVATE motoai-core)
    add_test(NAME runtime-lifecycle COMMAND runtime-test ${CMAKE_CURRENT_SOURCE_DIR}/../models/stories260K.gguf)
else()
    add_executable(motoai-reference EXCLUDE_FROM_ALL tools/runtime-run.cpp)
    target_link_libraries(motoai-reference PRIVATE motoai-core)
endif()
add_executable(motoai-chat-parse-replay EXCLUDE_FROM_ALL tools/chat-parse-replay.cpp)
target_link_libraries(motoai-chat-parse-replay PRIVATE motoai-core)
